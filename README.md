use parvez;

CREATE TABLE Users (
    user_id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(50),
    age INT,
    gender VARCHAR(10),
    country VARCHAR(50),
    signup_date DATE
);

CREATE TABLE Movies (
    movie_id INT PRIMARY KEY AUTO_INCREMENT,
    title VARCHAR(100),
    genre VARCHAR(50),
    release_year INT,
    rating FLOAT
);

CREATE TABLE Subscriptions (
    sub_id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT,
    plan VARCHAR(20),
    start_date DATE,
    end_date DATE,
    price DECIMAL(10,2),
    FOREIGN KEY (user_id) REFERENCES Users(user_id)
);

CREATE TABLE WatchHistory (
    watch_id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT,
    movie_id INT,
    watch_date DATE,
    watch_duration INT,
    FOREIGN KEY (user_id) REFERENCES Users(user_id),
    FOREIGN KEY (movie_id) REFERENCES Movies(movie_id)
);

-- Total number of users*
select COUNT(*) AS total_users
FROM users;
-- This shows total number users netflix has
---------------------------------------------

-- users by country.
 select country, COUNT(*) AS users_by

FROM users
GROUP BY country
ORDER BY users_by desc;
-- This shows count of users registered in each country and results are sorted by country with most users--------------------------------------------

-- subscription revenue by plan.
 SELECT  plan, SUM(price) as revenue
FROM subscriptions
GROUP BY plan
ORDER BY revenue asc;
-- It shows revenue generated by each plan and it gives lowest to highest revenue
-----------------------------------------------

-- Most popular subscription plan.
 select plan, COUNT(*) as popular
FROM subscriptions
GROUP BY plan
ORDER BY popular desc
limit 1 ;

------------------------------------------------

-- Top 5 raed movies.

FROM movies
GROUP BY title
ORDER BY top_rated desc
